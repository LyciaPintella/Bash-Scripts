# Place the following lines in /home/jessica/.inputrc for the local user or in
# /etc/inputrc for systemwide changes.
nano /home/jessica/.inputrc
nano /etc/inputrc

$include /etc/inputrc
set completion-ignore-case On
set bell-style none

# Reverse search history:
"\C-r": ""
"\C-f": reverse-search-history

# Cycle reverse search history with PageUp and PageDown:
"\e[5~": ""
"\e[6~": ""
"\e[5~": history-search-backward
"\e[6~": history-search-forward

# Cycle reverse search history with Up and Down:
"^[[A": ""
"^[[B": ""
"^[[A": history-search-backward
"^[[B": history-search-forward

# "^[[A" - up
# "^[[B" - down
Save it, then run:
bind -f ~/.inputrc
